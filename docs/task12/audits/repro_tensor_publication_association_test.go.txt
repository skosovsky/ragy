//go:build darwin || linux
package main
import("testing";"context";"github.com/skosovsky/ragy/access";"github.com/skosovsky/ragy/evidence";"github.com/skosovsky/ragy/retrieval";"github.com/skosovsky/ragy/source")
func TestAuditTensorRecorderUsesAllExecutedPublicationTargets(t *testing.T){
 // Arrange: use actual persistent executed stages, then the same consumer capture.
 p,stages,_:=recordedTensorStages(t);admitted,allowed:=tensorEvidencePolicyInputs(p.tensorRead,stages)
 input:=evidence.Input[meta]{Schema:p.tensor.Schema(),Codec:retrieval.NewJSONCodec[meta](p.tensor.Schema()),RetrievalID:"tensor-record",RecipeRevision:"saved-vectors",Outcome:evidence.Complete,Reason:evidence.NoReason,Coverage:retrieval.CompleteReadCoverage(),Required:[]evidence.Field{evidence.SourceField,evidence.ScoreField,evidence.PublicationField},Stages:stages,SourceAdmission:func(ctx context.Context,r access.Binding,ref source.Reference)error{return r.Check(ctx)}}
 policy:=evidence.Policy{AllowIdentifier:func(_ evidence.IdentifierKind,s string)bool{return allowed[s]},AllowNumbers:true,AllowLocation:func(l source.Locator)bool{return admitted[l.Reference]}}
 // Act.
 record,err:=evidence.Capture(t.Context(),p.tensorRead,input,policy);if err!=nil{t.Fatal(err)};snapshot,err:=record.Snapshot();if err!=nil{t.Fatal(err)}
 // Assert: one complete cross-stage snapshot must include both actual targets.
 hasDense:=false;for _,target:=range p.tensorRead.Publication().Targets(){if target.Target=="dense"{hasDense=true}}
 if !hasDense{t.Fatalf("complete record publication %s omits executed dense snapshot %s; stages=%d",*snapshot.Publication.Value,p.denseRead.Publication().Reference(),len(snapshot.Stages))}
}
