package lexical_test
import("context";"errors";"testing";"math";"time";ragy "github.com/skosovsky/ragy";"github.com/skosovsky/ragy/access";"github.com/skosovsky/ragy/filter";"github.com/skosovsky/ragy/lexical";"github.com/skosovsky/ragy/retrieval")
type auditNestedMeta struct{Tenant string `json:"tenant"`;Nested map[string][]string `json:"-"`}
func TestAuditSnapshotNestedMetadataOwned(t *testing.T){
 // Arrange.
 schema,read:=newSnapshotRead(t);original:=&auditNestedMeta{Tenant:"a",Nested:map[string][]string{"key":{"retained"}}}
 snap,err:=lexical.NewBM25Snapshot(t.Context(),schema,lexical.Config[*auditNestedMeta]{SearchFields:[]string{"content"}},read,[]retrieval.Document[*auditNestedMeta]{{ID:"d",Content:"secret",Meta:original}},func(m *auditNestedMeta)(*auditNestedMeta,error){out:=&auditNestedMeta{Tenant:m.Tenant,Nested:map[string][]string{}};for k,v:=range m.Nested{out.Nested[k]=append([]string(nil),v...)};return out,nil});if err!=nil{t.Fatal(err)}
 original.Nested["key"][0]="caller-mutated";q:=retrieval.Query[struct{}]{Read:read,Text:"secret",Options:retrieval.RetrieveOptions{TopK:1}};first,err:=snap.Retrieve(t.Context(),q);if err!=nil{t.Fatal(err)}
 // Act.
 first.Documents()[0].Meta.Nested["key"][0]="result-mutated";first.Documents()[0].Meta.Tenant="b";second,err:=snap.Retrieve(t.Context(),q)
 // Assert.
 if err!=nil||second.Len()!=1||second.Documents()[0].Meta.Nested["key"][0]!="retained"||second.Documents()[0].Meta.Tenant!="a"{t.Fatal("nested metadata aliases retained corpus",err)}
}
func TestAuditSnapshotCloneFailureStillChecksCancellation(t *testing.T){
 // Arrange: clone failure races with mandatory freshness gate.
 schema,read:=newSnapshotRead(t);ctx,cancel:=context.WithCancel(t.Context());defer cancel();calls:=0
 snap,err:=lexical.NewBM25Snapshot(t.Context(),schema,lexical.Config[snapshotMeta]{SearchFields:[]string{"content"}},read,[]retrieval.Document[snapshotMeta]{{ID:"d",Content:"secret",Meta:snapshotMeta{Tenant:"a"}}},func(m snapshotMeta)(snapshotMeta,error){calls++;if calls==2{cancel();return snapshotMeta{},errors.New("private-clone-error")};return m,nil});if err!=nil{t.Fatal(err)}
 // Act.
 rs,err:=snap.Retrieve(ctx,retrieval.Query[struct{}]{Read:read,Text:"secret",Options:retrieval.RetrieveOptions{TopK:1}})
 // Assert.
 if rs.Len()!=0||!access.IsProtectionFailure(err)||!errors.Is(err,context.Canceled){t.Fatalf("clone failure bypasses freshness: len=%d err=%v protected=%v",rs.Len(),err,access.IsProtectionFailure(err))}
}
func TestAuditFiniteHugeBM25FailsClosed(t *testing.T){
 // Arrange.
 idx,err:=lexical.NewBM25Index(filter.EmptySchema(),lexical.Config[struct{}]{SearchFields:[]string{"content"},K1:math.MaxFloat64,B:1},nil,nil);if err!=nil{t.Fatal(err)};if err=idx.Index([]retrieval.Document[struct{}]{{ID:"d",Content:"hello hello hello"}});err!=nil{t.Fatal(err)}
 // Act.
 rs,err:=idx.Retrieve(t.Context(),retrieval.Query[struct{}]{Read:retrieval.UnrestrictedRead(),Text:"hello",Options:retrieval.RetrieveOptions{TopK:1}})
 // Assert.
 if rs.Len()!=0||!errors.Is(err,ragy.ErrProtocol){t.Fatal("overflow delivered",err)}
}
func TestAuditSnapshotCaptureCloneFailureStillChecksCancellation(t *testing.T){
 // Arrange.
 schema,read:=newSnapshotRead(t);ctx,cancel:=context.WithCancel(t.Context());defer cancel()
 // Act: fail during initial capture cloning, after revoking the caller context.
 snap,err:=lexical.NewBM25Snapshot(ctx,schema,lexical.Config[snapshotMeta]{SearchFields:[]string{"content"}},read,[]retrieval.Document[snapshotMeta]{{ID:"d",Content:"secret",Meta:snapshotMeta{Tenant:"a"}}},func(m snapshotMeta)(snapshotMeta,error){cancel();return snapshotMeta{},errors.New("private-capture-clone-error")})
 // Assert.
 if snap!=nil||!access.IsProtectionFailure(err)||!errors.Is(err,context.Canceled){t.Fatalf("capture clone failure bypasses freshness: err=%v protected=%v",err,access.IsProtectionFailure(err))}
}
func TestAuditSnapshotCloneFailureStillChecksRevocation(t *testing.T){for _,phase:=range []string{"capture","delivery"}{t.Run(phase,func(t *testing.T){
 // Arrange: independently mutable trusted authority; callback failure is ordinary.
 schema,initial:=newSnapshotRead(t);mandatory,err:=initial.Prepare(t.Context(),schema,filter.Condition{},access.Capabilities{ScopeProfile:true,PinnedPublication:true,RequirePinnedPublication:true});if err!=nil{t.Fatal(err)};revoked:=false
 read,err:=access.Scoped(access.ScopedConfig{Schema:schema,Mandatory:mandatory,Publication:initial.Publication(),Snapshot:initial.Snapshot(),Now:func()time.Time{return initial.Snapshot().IssuedAt},Authority:access.AuthorityFunc(func(context.Context,access.Snapshot)error{if revoked{return ragy.ErrUnavailable};return nil})});if err!=nil{t.Fatal(err)}
 calls:=0;clone:=func(m snapshotMeta)(snapshotMeta,error){calls++;if phase=="capture"||calls==2{revoked=true;return snapshotMeta{},errors.New("sensitive-clone-error")};return m,nil}
 // Act.
 snap,err:=lexical.NewBM25Snapshot(t.Context(),schema,lexical.Config[snapshotMeta]{SearchFields:[]string{"content"}},read,[]retrieval.Document[snapshotMeta]{{ID:"d",Content:"secret",Meta:snapshotMeta{Tenant:"a"}}},clone)
 if phase=="delivery"{if err!=nil{t.Fatal(err)};rs,callErr:=snap.Retrieve(t.Context(),retrieval.Query[struct{}]{Read:read,Text:"secret",Options:retrieval.RetrieveOptions{TopK:1}});if rs.Len()!=0{t.Fatal("revoked data delivered")};err=callErr}
 // Assert.
 if !access.IsProtectionFailure(err)||!errors.Is(err,ragy.ErrUnavailable){t.Fatalf("revocation lost: %v",err)}
})}}
