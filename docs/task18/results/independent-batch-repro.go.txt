package correctnessrepro

import (
 "errors"
 "testing"
 "github.com/skosovsky/ragy/lifecycle"
 "github.com/skosovsky/ragy/lifecycle/filestore"
 "github.com/skosovsky/ragy/source"
)

func batch() lifecycle.Snapshot {
 ref := source.Reference{Namespace:"n",Source:"s",Revision:"r",Transformation:"t",AccessFingerprint:"a",Artifact:"one",Representation:"text"}
 makePlan := func(id, content, payload string) lifecycle.Manifest {
  return lifecycle.Manifest{ID:id,Key:id,Payload:payload,State:lifecycle.Planned,Identity:lifecycle.Identity{Namespace:"n",Source:"s",Revision:"r",Content:content,Transformation:"t",Access:"a"},Targets:[]lifecycle.Target{{Name:"dense",Required:true,State:lifecycle.TargetPending,Artifacts:[]lifecycle.Artifact{{Reference:ref,Supports:[]source.Reference{ref}}}}}}
 }
 return lifecycle.Snapshot{Schema:lifecycle.SchemaIdentity,Namespace:"n",Manifests:[]lifecycle.Manifest{makePlan("first","content-one","payload-one"),makePlan("second","content-two","payload-two")}}
}

func TestRawCASBatchCannotAliasExactArtifact(t *testing.T) {
 // Arrange: two new, different payload plans reserve the same target/reference.
 next := batch()
 current := lifecycle.Snapshot{Schema:lifecycle.SchemaIdentity,Namespace:"n"}
 store, err := filestore.New(t.TempDir(),1<<20)
 if err != nil { t.Fatal(err) }
 // Act: exercise the custom-Store contract and actual durable Store path.
 replacementErr := lifecycle.ValidateReplacement(current,next)
 committed, casErr := store.CompareSwap(t.Context(),0,next)
 retained, loadErr := store.Load(t.Context(),"n")
 // Assert: no conflicting reservation reaches durable storage.
 t.Logf("snapshot.Validate=%v replacementErr=%v casErr=%v returnedGeneration=%d retainedGeneration=%d retainedManifests=%d loadErr=%v",next.Validate(),replacementErr,casErr,committed.Generation,retained.Generation,len(retained.Manifests),loadErr)
 if !errors.Is(replacementErr,lifecycle.ErrConflict) || !errors.Is(casErr,lifecycle.ErrConflict) || loadErr != nil || retained.Generation != 0 || len(retained.Manifests)!=0 {
  t.Fatal("raw CAS admitted distinct payload plans sharing exact target/artifact identity")
 }
}
